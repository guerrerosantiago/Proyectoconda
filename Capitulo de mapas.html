# Agregar marcadores de ciudades principales
ciudades = {
    'Bogotá': [4.6097, -74.0817],
    'Medellín': [6.2442, -75.5812],
    'Cali': [3.4516, -76.5320],
    'Barranquilla': [10.9685, -74.7813],
    'Cartagena': [10.3910, -75.4794]
}

for ciudad, coords in ciudades.items():
    folium.Marker(
        coords,
        popup=ciudad,
        tooltip=ciudad,
        icon=folium.Icon(color='red', icon='info-sign')
    ).add_to(m)

import folium

# Mapa básico de Colombia
m = folium.Map(location=[4.6, -74.08], zoom_start=6)

# Agregar marcadores de ciudades principales con colores diferentes
ciudades = {
    'Bogotá': ([4.6097, -74.0817], 'red'),
    'Medellín': ([6.2442, -75.5812], 'green'),
    'Cali': ([3.4516, -76.5320], 'blue'),
    'Barranquilla': ([10.9685, -74.7813], 'purple'),
    'Cartagena': ([10.3910, -75.4794], 'orange'),
    'Santa Marta': ([11.2408, -74.1990], 'darkblue'),
    'Bucaramanga': ([7.1254, -73.1198], 'darkred')
}

for ciudad, (coords, color) in ciudades.items():
    folium.Marker(
        coords,
        popup=ciudad,
        tooltip=ciudad,
        icon=folium.Icon(color=color, icon='info-sign')
    ).add_to(m)
